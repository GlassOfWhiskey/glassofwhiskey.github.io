{{- /* fmt-text.html — apply typographic replacements to plain text. Replaces: ^text^ → text (superscript) --- → — (em dash) -- → – (en dash) Usage: {{ partial "fmt-text.html" .someField | safeHTML }} */ -}} {{- $s := . -}} {{- $s = $s | replaceRE "``([^`]+)``" "$1" -}} {{- $s = $s | replaceRE "`([^`]+)`" "$1" -}} {{- $s = $s | replaceRE `\^([^^]+)\^` "$1" -}} {{- $s = $s | replaceRE "---" "—" -}} {{- $s = $s | replaceRE "--" "–" -}} {{- return $s -}}
